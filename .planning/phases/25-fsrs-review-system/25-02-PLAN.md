  **Method 1: `get_review_overview(player_id: str) -> list[dict]`**

  Returns due review counts per subject for a player.

  ```python
  @frappe.whitelist(allow_guest=False)
  def get_review_overview(player_id: str) -> list[dict]:
      """Get count of due reviews per subject for a player.

      Uses composite index on (player, subject, next_review).
      Returns: [{"subject": "SUBJ-00001", "due_count": 15}, ...]
      """
      today = frappe.utils.today()  # Returns 'YYYY-MM-DD' string
      return frappe.db.sql("""
          SELECT subject, COUNT(*) as due_count
          FROM `tabMemora Memory State`
          WHERE player = %(player)s
            AND next_review <= %(today)s
          GROUP BY subject
      """, {"player": player_id, "today": today}, as_dict=True)
  ```

  NOTE: Since `next_review` is always midnight (clamped in Plan 01), `next_review <= '2026-02-09'` correctly matches stages due today (their next_review is '2026-02-09 00:00:00' which is <= '2026-02-09'). For stages due yesterday, '2026-02-08 00:00:00' <= '2026-02-09' is also true. This works correctly.

  **Method 2: `get_due_stages(player_id: str, subject_id: str, limit: int = 10) -> list[dict]`**

  Returns up to `limit` due stages for a specific subject, oldest first (FIFO). Validates stages still exist in their lessons (handles removed stages per SC #10).

  ```python
  @frappe.whitelist(allow_guest=False)
  def get_due_stages(player_id: str, subject_id: str, limit: int = 10) -> list[dict]:
      """Get up to N due stages for a subject, oldest first (FIFO).

      Returns stage_id, lesson_id, stage_type, and memory_state_name.
      Validates stages still exist in their lessons (skips removed stages).
      Over-fetches by 5 to account for removed stages.
      """
      limit = int(limit)  # Frappe passes strings from API calls
      today = frappe.utils.today()

      rows = frappe.db.sql("""
          SELECT ms.name as memory_state_name, ms.stage_id, ms.lesson,
                 ms.stability, ms.difficulty, ms.next_review
          FROM `tabMemora Memory State` ms
          WHERE ms.player = %(player)s
            AND ms.subject = %(subject)s
            AND ms.next_review <= %(today)s
          ORDER BY ms.next_review ASC
          LIMIT %(fetch_limit)s
      """, {
          "player": player_id,
          "subject": subject_id,
          "today": today,
          "fetch_limit": limit + 5,  # Over-fetch to account for removed stages
      }, as_dict=True)

      result = []
      for row in rows:
          if len(result) >= limit:
              break

          # Validate stage still exists in lesson and get stage_type
          stage_info = frappe.db.get_value(
              "Memora Lesson Stage",
              {"parent": row.lesson, "stage_title": row.stage_id},
              ["stage_type"],
              as_dict=True,
          )

          if stage_info:
              result.append({
                  "stage_id": row.stage_id,
                  "lesson_id": row.lesson,
                  "stage_type": stage_info.stage_type,
                  "memory_state_name": row.memory_state_name,
                  "stability": row.stability,
                  "difficulty": row.difficulty,
              })
          # else: stage was removed from lesson, skip gracefully (SC #10)

      # Calculate has_more: are there more due stages beyond what we returned?
      total_due = frappe.db.count("Memora Memory State", {
          "player": player_id,
          "subject": subject_id,
          "next_review": ["<=", today],
      })
      has_more = total_due > len(result)

      return {"stages": result, "has_more": has_more}
  ```

  **Method 3: `submit_reviews(player_id: str, subject_id: str, stages: str) -> dict`**

  Accepts batch of reviewed stages (as JSON string since Frappe passes strings), runs inline FSRS, updates Memory State records, returns remaining count.

  ```python
  import json
  from datetime import date, datetime, time, timedelta, timezone

  @frappe.whitelist(allow_guest=False)
  def submit_reviews(player_id: str, subject_id: str, stages: str) -> dict:
      """Submit batch of reviewed stages and update Memory States via FSRS.

      Args:
          player_id: Player profile name
          subject_id: Subject ID
          stages: JSON string of list[{"stage_id": str, "fail_count": int}]

      Returns:
          {"processed": int, "remaining_due": int, "has_more": bool}
      """
      from fsrs import Card, Rating, Scheduler

      # Parse stages from JSON string (Frappe API passes strings)
      if isinstance(stages, str):
          stages_list = json.loads(stages)
      else:
          stages_list = stages

      # Get FSRS scheduler (reuse logic from fsrs_processor)
      scheduler = _get_fsrs_scheduler()

      processed = 0
      now = datetime.now(timezone.utc)

      for stage_data in stages_list:
          stage_id = stage_data["stage_id"]
          fail_count = stage_data.get("fail_count", 0)

          # Find the Memory State record for this player+subject+stage
          # Name format: {season}-{subject}-{player}-{stage_id}
          memory_state = frappe.db.sql("""
              SELECT name, stability, difficulty, next_review
              FROM `tabMemora Memory State`
              WHERE player = %(player)s
                AND subject = %(subject)s
                AND stage_id = %(stage_id)s
              LIMIT 1
          """, {
              "player": player_id,
              "subject": subject_id,
              "stage_id": stage_id,
          }, as_dict=True)

          if not memory_state:
              continue  # Stage not found, skip

          ms = memory_state[0]

          # Reconstruct FSRS card
          card = Card()
          card.stability = ms.stability or 0
          card.difficulty = ms.difficulty or 0
          card.due = ms.next_review if ms.next_review else now

          # Map fail_count to FSRS rating
          if fail_count == 0:
              rating = Rating.Good
          elif fail_count == 1:
              rating = Rating.Hard
          else:
              rating = Rating.Again

          # Run FSRS review
          card, _log = scheduler.review_card(card, rating, now)

          # Clamp next_review to midnight, minimum tomorrow
          next_date = card.due.date()
          tomorrow = date.today() + timedelta(days=1)
          if next_date < tomorrow:
              next_date = tomorrow
          next_review_naive = datetime.combine(next_date, time.min)

          # Update Memory State record
          frappe.db.set_value(
              "Memora Memory State",
              ms.name,
              {
                  "stability": card.stability,
                  "difficulty": card.difficulty,
                  "next_review": next_review_naive,
              },
              update_modified=True,
          )

          processed += 1

      # Commit all updates
      if processed > 0:
          frappe.db.commit()

      # Calculate remaining due
      today = frappe.utils.today()
      remaining_due = frappe.db.count("Memora Memory State", {
          "player": player_id,
          "subject": subject_id,
          "next_review": ["<=", today],
      })

      return {
          "processed": processed,
          "remaining_due": remaining_due,
          "has_more": remaining_due > 0,
      }
  ```

  **Helper function** (reuse from fsrs_processor pattern):

  ```python
  def _get_fsrs_scheduler():
      """Create FSRS scheduler with weights from Memora Settings."""
      from fsrs import Scheduler

      settings = frappe.get_single("Memora Settings")
      weights_str = settings.fsrs_weights

      if weights_str and weights_str.strip():
          try:
              weights = json.loads(weights_str)
              return Scheduler(parameters=weights)
          except (json.JSONDecodeError, ValueError, TypeError):
              pass

      return Scheduler()
  ```
